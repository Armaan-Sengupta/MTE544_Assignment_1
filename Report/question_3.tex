\section{Question 3: 3-Omniwheel Simulation}

\begin{enumerate}[label=\textbf{(\alph*)}]
    \item To derive the equations of motion for the 3-wheel omnidirectional robot, we define the coordinate frames and geometric parameters as shown in Figure~\ref{fig:q3_frame}.

    \begin{center}
        \includegraphics[width=0.48\linewidth]{img/q3_frame.png}
        \captionof{figure}{3-Omniwheel robot frame assignments and wheel configuration.}
        \label{fig:q3_frame}
    \end{center}

    From the lecture derivation for omnidirectional wheeled mobile bases, the rotational speed $u_i$ of wheel $i$ is related to the robot velocity $\dot{q} = \begin{bmatrix} \dot{x} & \dot{y} & \dot{\theta} \end{bmatrix}^\top$ by:
    \begin{equation}
        u_i = \frac{1}{r_i} \begin{bmatrix} 1 & \tan\gamma_i \end{bmatrix} g_i(\theta)\,\dot{q}
    \end{equation}
    where $g_i(\theta)$ is
    \begin{equation}
        g_i(\theta) = \begin{bmatrix} 
            \cos(\theta + \beta_i) & \sin(\theta + \beta_i) & X_i \sin\beta_i - Y_i \cos\beta_i \\ 
            -\sin(\theta + \beta_i) & \cos(\theta + \beta_i) & X_i \cos\beta_i + Y_i \sin\beta_i 
        \end{bmatrix}
    \end{equation}

    For our symmetrical 3-omniwheel configuration:
    \begin{itemize}
        \item All three wheels share identical radius: $r = r_1 = r_2 = r_3$.
        \item The roller angle for standard omniwheels is perpendicular to the wheel plane: $\gamma = \gamma_1 = \gamma_2 = \gamma_3 = 0^\circ$.
        \item The wheel orientation angles (measured from the positive x-axis of the robot frame to the positive x-axis of the wheel frame, shown in Figure~\ref{fig:q3_frame}) are separated by $120^\circ$:
        \begin{equation*}
            \beta_1 = 90^\circ, \quad \beta_2 = 210^\circ, \quad \beta_3 = 330^\circ
        \end{equation*}
    \end{itemize}

    Since $\tan\gamma_i = \tan(0^\circ) = 0$, only the first row of $g_i(\theta)$ contributes to driving the wheel:
    \begin{align}
        u_i &= \frac{1}{r} \begin{bmatrix} 1 & 0 \end{bmatrix} g_i(\theta)\,\dot{q} \notag \\
        &= \frac{1}{r} \Big( \dot{x}\cos(\theta + \beta_i) + \dot{y}\sin(\theta + \beta_i) + \dot{\theta}\big(X_i\sin\beta_i - Y_i\cos\beta_i\big) \Big) \label{eq:ui_general}
    \end{align}

    Each wheel is mounted at a distance $L$ from the robot center. Let $\alpha_i$ be the angle between the positive x-axis of the robot frame and the line connecting the robot center to the center of wheel, which is to say the angle to $Y_i$, where $\alpha_i = \beta_i - 90^\circ$. The wheel center coordinates $(X_i, Y_i)$ are:
    \begin{align*}
        X_i &= L\cos\alpha_i = L\cos(\beta_i - 90^\circ) = L\sin\beta_i \\
        Y_i &= L\sin\alpha_i = L\sin(\beta_i - 90^\circ) = -L\cos\beta_i
    \end{align*}
    Substituting $(X_i, Y_i)$ into the coefficient of $\dot{\theta}$:
    \begin{align}
        X_i \sin\beta_i - Y_i \cos\beta_i &= (L\sin\beta_i)\sin\beta_i - (-L\cos\beta_i)\cos\beta_i \notag \\
        &= L\big(\sin^2\beta_i + \cos^2\beta_i\big) \notag \\
        &= L \label{eq:lever_arm}
    \end{align}
    This neatly simplifies to the constant distance $L$ for every wheel, which makes sense intuitively because all wheels are symmetrically located at distance $L$ from the center.

    \paragraph{Individual Wheel Equations:}
    Substituting \eqref{eq:lever_arm} back into \eqref{eq:ui_general} gives the general relation:
    \begin{equation}
        u_i = \frac{1}{r} \Big( \dot{x}\cos(\theta + \beta_i) + \dot{y}\sin(\theta + \beta_i) + L\dot{\theta} \Big)
    \end{equation}

    Plugging in the angles $\beta_1 = 90^\circ$, $\beta_2 = 210^\circ$, $\beta_3 = 330^\circ$ for each wheel:
    \begin{align}
        u_1 &= \frac{1}{r}\Big( \dot{x}\cos(\theta + 90^\circ) + \dot{y}\sin(\theta + 90^\circ) + L\dot{\theta} \Big) \\
        u_2 &= \frac{1}{r}\Big( \dot{x}\cos(\theta + 210^\circ) + \dot{y}\sin(\theta + 210^\circ) + L\dot{\theta} \Big) \\
        u_3 &= \frac{1}{r}\Big( \dot{x}\cos(\theta + 330^\circ) + \dot{y}\sin(\theta + 330^\circ) + L\dot{\theta} \Big)
    \end{align}

    Collecting all three equations into matrix form yields:
    \begin{equation}
        \begin{bmatrix} u_1 \\ u_2 \\ u_3 \end{bmatrix} = \frac{1}{r} G \begin{bmatrix} \dot{x} \\ \dot{y} \\ \dot{\theta} \end{bmatrix} = \frac{1}{r} \begin{bmatrix}
            \cos(\theta + 90^\circ) & \sin(\theta + 90^\circ) & L \\
            \cos(\theta + 210^\circ) & \sin(\theta + 210^\circ) & L \\
            \cos(\theta + 330^\circ) & \sin(\theta + 330^\circ) & L
        \end{bmatrix} \begin{bmatrix} \dot{x} \\ \dot{y} \\ \dot{\theta} \end{bmatrix}
    \end{equation}

    \item To simulate the robot, we need the state derivatives $\dot{q} = \begin{bmatrix} \dot{x} & \dot{y} & \dot{\theta} \end{bmatrix}^\top$ as functions of the wheel speeds $u = \begin{bmatrix} u_1 & u_2 & u_3 \end{bmatrix}^\top$. Inverting the forward kinematic relation $\begin{bmatrix} u_1 \\ u_2 \\ u_3 \end{bmatrix} = \frac{1}{r} G \begin{bmatrix} \dot{x} \\ \dot{y} \\ \dot{\theta} \end{bmatrix}$ yields:
    \begin{equation}
        \begin{bmatrix} \dot{x} \\ \dot{y} \\ \dot{\theta} \end{bmatrix} = r G^{-1} \begin{bmatrix} u_1 \\ u_2 \\ u_3 \end{bmatrix}
    \end{equation}
    Inverting $G$ symbolically in MATLAB gives the inverse kinematic matrix:
    \begin{equation}
        \begin{bmatrix} \dot{x} \\ \dot{y} \\ \dot{\theta} \end{bmatrix} = \frac{r}{3} \begin{bmatrix}
            -2\sin\theta & -2\cos\left(\theta + \frac{\pi}{6}\right) & 2\sin\left(\theta + \frac{\pi}{3}\right) \\[0.5em]
            2\cos\theta & -2\cos\left(\theta - \frac{\pi}{3}\right) & -2\cos\left(\theta + \frac{\pi}{3}\right) \\[0.5em]
            \frac{1}{L} & \frac{1}{L} & \frac{1}{L}
        \end{bmatrix} \begin{bmatrix} u_1 \\ u_2 \\ u_3 \end{bmatrix}
    \end{equation}

    \begin{enumerate}[label=\Roman*.]
        \item \textbf{Rotation Inputs:} $u_1 = -2.0\,\text{rad/s}$, $u_2 = 1.0\,\text{rad/s}$, $u_3 = 1.0\,\text{rad/s}$.

        We simulate the true motion of the robot for $30\,\text{s}$ with a timestep of $\Delta t = 0.1\,\text{s}$, using $r = 0.10\,\text{m}$ and $L = 0.25\,\text{m}$ starting from $(x, y, \theta) = (0, 0, 0)$.

        \begin{center}
            \includegraphics[width=0.85\linewidth]{img/q3_case1.png}
            \captionof{figure}{Trajectory and state variables for Part (b).I ($u_1 = -2\,\text{rad/s}, u_2 = 1\,\text{rad/s}, u_3 = 1\,\text{rad/s}$).}
        \end{center}

        \textbf{Observations:} Since the sum of wheel speeds is $u_1 + u_2 + u_3 = -2 + 1 + 1 = 0$, the net angular velocity $\dot{\theta} = \frac{r}{3L}(u_1 + u_2 + u_3) = 0\,\text{rad/s}$. The robot doesn't rotate at all, so its heading stays flat at $\theta(t) = 0\,\text{rad}$. 
        This makes intuative sense as we can imagine each of the three wheels canceling eachother out.

        Along the $y$-axis, all three wheels push together in the $-y$ direction to give a constant speed. As expected, the robot traces a straight line along the $-y$ axis.

        \item Designed control signals
        \begin{enumerate}[label=\arabic*.]
            \item Straight line with a slope of 60 degrees

            To move in a straight line with a slope of $60^\circ$, the trajectory must satisfy:
            \begin{equation}
                \frac{\dot{y}}{\dot{x}} = \tan(60^\circ) = \sqrt{3} \implies \dot{y} = \tan(60^\circ) \dot{x}
            \end{equation}
            Choosing a constant linear velocity along the $x$-axis of $\dot{x} = 1\,\text{m/s}$ yields:
            \begin{align}
                \dot{x} &= 1\,\text{m/s} \\
                \dot{y} &= \sqrt{3}\,\text{m/s} \approx 1.732\,\text{m/s} \\
                \dot{\theta} &= 0\,\text{rad/s}
            \end{align}
            Since the robot starts at orientation $\theta(0) = 0$ and $\dot{\theta} = 0$, the orientation remains $\theta(t) = 0$ for all time. Substituting $\theta = 0$, $\dot{q} = [1, \sqrt{3}, 0]^T$, and wheel radius $r = 0.10\,\text{m}$ into the forward kinematics relation $u = \frac{1}{r} G(0) \dot{q}$:
            \begin{equation}
                \begin{bmatrix} u_1 \\ u_2 \\ u_3 \end{bmatrix} = \frac{1}{r} \begin{bmatrix} -\sin(0) & \cos(0) & L \\ -\sin(\frac{2\pi}{3}) & \cos(\frac{2\pi}{3}) & L \\ -\sin(\frac{4\pi}{3}) & \cos(\frac{4\pi}{3}) & L \end{bmatrix} \begin{bmatrix} \dot{x} \\ \dot{y} \\ \dot{\theta} \end{bmatrix} = \frac{1}{0.10} \begin{bmatrix} 0 & 1 & L \\ -\frac{\sqrt{3}}{2} & -\frac{1}{2} & L \\ \frac{\sqrt{3}}{2} & -\frac{1}{2} & L \end{bmatrix} \begin{bmatrix} 1 \\ \sqrt{3} \\ 0 \end{bmatrix}
            \end{equation}
            Evaluating each wheel velocity:
            \begin{align}
                u_1 &= \frac{1}{0.10} (0 \cdot 1 + 1 \cdot \sqrt{3} + 0) = \frac{\sqrt{3}}{0.10} = 10\sqrt{3} \approx 17.32\,\text{rad/s} \\
                u_2 &= \frac{1}{0.10} \left(-\frac{\sqrt{3}}{2} \cdot 1 - \frac{1}{2} \cdot \sqrt{3} + 0\right) = \frac{-\sqrt{3}}{0.10} = -10\sqrt{3} \approx -17.32\,\text{rad/s} \\
                u_3 &= \frac{1}{0.10} \left(\frac{\sqrt{3}}{2} \cdot 1 - \frac{1}{2} \cdot \sqrt{3} + 0\right) = 0\,\text{rad/s}
            \end{align}

            The simulation is run for $t = 30\,\text{s}$ with initial state $q(0) = [0, 0, 0]^T$.

            \begin{center}
                \includegraphics[width=0.85\linewidth]{img/q3_case2.png}
                \captionof{figure}{Trajectory and state variables for $60^\circ$ slope straight line ($u_1 = 17.32\,\text{rad/s}, u_2 = -17.32\,\text{rad/s}, u_3 = 0\,\text{rad/s}$).}
            \end{center}

            \textbf{Observations:} With $u_1 = -u_2$ and $u_3 = 0$, the sum of wheel speeds is $u_1 + u_2 + u_3 = 10\sqrt{3} - 10\sqrt{3} + 0 = 0$, meaning the net rotational speed is exactly zero and the robot maintains $\theta(t) = 0$ throughout. 
            Wheel 3 does not spin at all ($u_3 = 0$) and simply rolls freely on its passive rollers as wheels 1 and 2 drive the robot diagonally. 
            Over 30 seconds, the robot reaches $(x, y) = (30.00, 51.96)\,\text{m}$, maintaining the exact commanded slope of $51.96 / 30.00 = 1.732 = \tan(60^\circ)$.

            \break
            
            \item Two-metre diameter circle

            To trace a circle with a diameter of $D = 2\,\text{m}$, the path radius is $R = 1\,\text{m}$. As illustrated in Figure~\ref{fig:q3_circle_icr}, we define the instantaneous center of rotation (ICR) such that the robot body frame $\{R\}$ has its forward axis $x_R$ tangent to the trajectory with forward linear speed $v$, while rotating about the ICR with angular velocity $\omega = \dot{\theta}$.

            \begin{center}
                \includegraphics[width=0.48\linewidth]{img/q3_circle_icr.png}
                \captionof{figure}{Robot body frame $\{R\}$ aligned tangentially along the circular trajectory of radius $R$ about the ICR.}
                \label{fig:q3_circle_icr}
            \end{center}

            The relationship between linear velocity and angular velocity for circular motion is:
            \begin{equation}
                v = \omega R
            \end{equation}
            With $R = 1\,\text{m}$, this simplifies to $v = \omega$. Choosing an angular velocity of $\omega = 1\,\text{rad/s} = \dot{\theta}$ yields a linear velocity of $v = 1\,\text{m/s}$.

            In the robot body frame $\{R\}$, the velocity components are:
            \begin{align}
                \dot{x}^R &= v = 1\,\text{m/s} \\
                \dot{y}^R &= 0\,\text{m/s} \\
                \dot{\theta}^R &= \dot{\theta} = 1\,\text{rad/s}
            \end{align}

            To evaluate the forward kinematics, we transform the linear velocity from the robot frame $\{R\}$ to the global frame $\{S\}$ via rotation matrix $R_R^S(\theta)$:
            \begin{equation}
                v^S = R_R^S(\theta) v^R = \begin{bmatrix} \cos\theta & -\sin\theta \\ \sin\theta & \cos\theta \end{bmatrix} \begin{bmatrix} 1 \\ 0 \end{bmatrix} = \begin{bmatrix} \cos\theta \\ \sin\theta \end{bmatrix} = \begin{bmatrix} \dot{x} \\ \dot{y} \end{bmatrix}
            \end{equation}
            \begin{equation}
                \dot{q} = \begin{bmatrix} \dot{x} \\ \dot{y} \\ \dot{\theta} \end{bmatrix} = \begin{bmatrix} \cos\theta \\ \sin\theta \\ 1 \end{bmatrix}
            \end{equation}

            Substituting $\dot{q}$ into the forward kinematics relation $u = \frac{1}{r} G(\theta) \dot{q}$:
            \begin{equation}
                \begin{bmatrix} u_1 \\ u_2 \\ u_3 \end{bmatrix} = \frac{1}{r} \begin{bmatrix}
                    \cos(\theta + 90^\circ) & \sin(\theta + 90^\circ) & L \\
                    \cos(\theta + 210^\circ) & \sin(\theta + 210^\circ) & L \\
                    \cos(\theta + 330^\circ) & \sin(\theta + 330^\circ) & L
                \end{bmatrix} \begin{bmatrix} \cos\theta \\ \sin\theta \\ 1 \end{bmatrix}
            \end{equation}
            Simplifying (via MATLAB) the orientation angle $\theta$ cancels out entirely, yielding constant wheel speeds:
            \begin{equation}
                \begin{bmatrix} u_1 \\ u_2 \\ u_3 \end{bmatrix} = \frac{1}{r} \begin{bmatrix} L \\ L - \frac{\sqrt{3}}{2} \\ L + \frac{\sqrt{3}}{2} \end{bmatrix}
            \end{equation}

            Substituting the robot dimensions $r = 0.10\,\text{m}$ and $L = 0.25\,\text{m}$:
            \begin{align}
                u_1 &= \frac{0.25}{0.10} = 2.5\,\text{rad/s} \\
                u_2 &= \frac{0.25 - \frac{\sqrt{3}}{2}}{0.10} = 2.5 - 5\sqrt{3} \approx -6.16\,\text{rad/s} \\
                u_3 &= \frac{0.25 + \frac{\sqrt{3}}{2}}{0.10} = 2.5 + 5\sqrt{3} \approx 11.16\,\text{rad/s}
            \end{align}

            The simulation is executed for $t = 30\,\text{s}$ starting from $q(0) = [0, 0, 0]^T$ with timestep $\Delta t = 0.1\,\text{s}$.

            \begin{center}
                \includegraphics[width=0.85\linewidth]{img/q3_case3.png}
                \captionof{figure}{Trajectory and state variables for $2\,\text{m}$ diameter circle ($u_1 = 2.5\,\text{rad/s}, u_2 = -6.16\,\text{rad/s}, u_3 = 11.16\,\text{rad/s}$).}
            \end{center}

            \textbf{Observations:} Even though the robot traverses a curved circular path, the required wheel inputs $u_1, u_2, u_3$ are completely constant! This occurs because the robot's heading $\theta(t)$ rotates at the exact same rate as its tangent velocity vector ($\dot{\theta} = \omega = 1\,\text{rad/s}$), keeping the relative velocity of the ground in the robot's local frame invariant.
            This is more intuative with a two wheel robot when one wheel is spinning a fixed rate faster than the other, but the same can be acomplished with our 3 omniwheel robot.
            
            Because of the $\omega$ / $v$ we choose as 1, it completes one full revolution every $2\pi \approx 6.28\,\text{s}$. In the Cartesian plane, $x(t)$ and $y(t)$ oscillate sinusoidally with amplitude $1.0\,\text{m}$, creating a closed circular path of diameter $2.0\,\text{m}$ centered at $(0, 1)\,\text{m}$. Over the 30-second duration, the robot traces approximately $4.77$ ($\frac{30}{2\pi}$) full laps around the circle.
        \end{enumerate}
    \end{enumerate}
\end{enumerate}
